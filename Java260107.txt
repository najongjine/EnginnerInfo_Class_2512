package com.javatest;

abstract class Shape {
	int a=1;
	abstract void draw();
}

class Bread260107 extends Shape{
	String flourkind="밀가루";
	int flourgram=10;
	
	Bread260107(String flourkind){
		System.out.println("빵 클래스");
	}

	
	void draw() {
		System.out.println("빵모양");
		
	}
}
                                                                
class FishBread260107 extends Bread260107{
	String flourkind="허니밀가루";
	FishBread260107(){
		super("흑가루");
		System.out.println("붕어빵 클래스");
	}
	FishBread260107(String flourkind){
		super(flourkind);
		this.flourkind=flourkind;
		super.flourkind="";
	}
	public void draw() {
		System.out.println("붕어빵모양");
	}
}

public class Java260107 {

	public static void main(String[] args) {
		// 상속
		/*
		 * 프로그래밍 세계에선 자식은 부모꺼 다 물려받음
		 * 부모가 먼저 생성되어야, 자식이 생성될수 있음
		 * 클래스이름() 이걸 기본 생성자라고 함. 클래스가 만들어질때 무조건
		 * 호출함.
		 * 상속 상태에서, 부모에 생성자를 단 한개로 안만들었다? 지가 내용물 없는 껍데기 생성자를
		 * 컴파일 단계에서 만듬.
		 * 
		 * 부모에 클래스이름() 이런 기본 생성자 만들어줬다? no problem. 그거 갔다 
		 * 쓰면 됨
		 * 
		 * 부모에 클래스이름(매개변수) 이런 생성자는 있지만, 기본 생성자가 없다?
		 * 난리 난다
		 * 해결 방법은, 부모에 기본 생성자도 만들어 주던가, 아니면
		 * 자식 클래스 생성자에서 부모 생성자를 강제로 호출 시켜야 한다
		 * 
		 * abstract(추상) class 는 미완성 클래스.
		 * 속성은 상관 없지만, method 는 껍데기만 정의할수 있어요.
		 * abstract 클래스를 사용하고 싶으면, 무조건 상속을 해야해요.
		 * 속성은 그냥 물려 주지만, 만약 method가 abstract class
		 * 에 정의됬다면, 자식 클래스에서 무조건 내용물 채워 줘야해요 
		 * 
		 * 타입이름이랑, 생성 이름이랑 같으면 머리 안아파요.
		 * 나한테 없는건, 부모한테서 갔다되고, 나한테 있는건 내꺼 쓰면 되요
		 * */
		FishBread260107 fishbread= 
				new FishBread260107("허니가루");
		fishbread.draw();
		fishbread.a=1;
		

	}

}
