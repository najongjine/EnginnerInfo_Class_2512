package com.javatest;


class FishBreadRecipe{
	// 속성, attribute
	int sweetBean=10;
	int flour=20;
}

public class Java260102 {

	public static void main(String[] args) {
		/*
		 * 자바는 포인터가 기호가 없지만, 포인터 원리가 없다는 뜻이 아니에요
		 * new 클래스명() 이렇게 하면 heap 메모리 영역에 데이터를 만들고,
		 * 그 데이터가 있는 주소를 fishBread 에 다가 저장한다
		 * */
		FishBreadRecipe fishBread=new FishBreadRecipe();
		FishBreadRecipe fishBread2=new FishBreadRecipe();
		FishBreadRecipe fishBread3=fishBread;
		if(fishBread==fishBread2) {
			System.out.println("fishBread 와 fishBread2 는 주소가 같아요");
		}
		if(fishBread==fishBread3) {
			System.out.println("fishBread 와 fishBread3 는 주소가 같아요");
		}
		if(fishBread2==fishBread3) {
			System.out.println("fishBread2 와 fishBread3 는 주소가 같아요");
		}
		
		/*
		 * 타입이 소문자다(int, boolean) - 그냥 값 복붙해옴
		 * 타입이 대문자다(Class A) - 주소 개념이다
		 * int a == int b 그냥 값 비교함 simple
		 * class A == class B  주소 비교함. 속사정이 복잡함
		 * 하지만 String 이 나온다면... 좀... 구렁이 담 넘어가듯 비리좀 저지르자
		 * String은 깐깐하게 하지말고 그냥 값 보여줘
		 * */
		String str1="abc";
		String str2="abc";
		String str3=new String("abc");
		System.out.println(System.identityHashCode(str1)); // 예: 123456
		System.out.println(System.identityHashCode(str2)); // 예: 123456 (str1과 같음!)
		System.out.println(System.identityHashCode(str3)); // 예: 789012 (str1과 다름!)
		
	}

}
