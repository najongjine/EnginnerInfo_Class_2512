package com.javatest;

class Bread{
	String flourKind="밀가루";
	int price=500;
	
	void showBread() {
		System.out.println(" 밀가루 종류: "+flourKind);
		System.out.println(" 가격: "+price);
	}
	int 가격사기치기() {
		return price*10;
	}
	
	/* String, float, boolean, int, char 얘네들  
	 * 값 복붙으로 전달되요.
	 * 원본은 손상 안가요
	 * */
	void 원자재사기치기(String mat) {
		flourKind=mat;
		mat="밀당성공";
	}
	void 가격사기치기2(int _price) {
		price=_price;
		_price=9999999;
	}
	
	/*
	 * call by refrence. 
	 * 주소를 넘겨줘서, 원본이 손상감
	 * */
	void 빵에장난치기(Bread _bread) {
		_bread.flourKind="흙";
	}
	Bread 빵에장난치기2(Bread _bread) {
		_bread.flourKind="흙";
		return _bread;
	}
}

public class Java260105 {

	public static void main(String[] args) {
		Bread bread = new Bread();
		Bread bread2 = new Bread(); // 001
		bread.빵에장난치기2(bread2);
		System.out.println("bread1 : "+bread.flourKind);
		System.out.println("bread2 : "+bread2.flourKind);
	}

}
