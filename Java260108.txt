package com.javatest;

import java.util.ArrayList;
import java.util.List;

//Bread260108 fishbread =new FishBread260108("꿀가루");
class Bread260108{
	public int count=1;
	void Show() {
		System.out.println("빵이에요");
	}
	void ShowCount() {
		System.out.println(count);
		
	}
	
}
class FishBread260108 extends Bread260108{
	
	String flourkind;
	FishBread260108(){
		this.flourkind="밀가루";
	}
	FishBread260108(String flourkind){
		this.flourkind=flourkind;
	}
	
	void Show() {
		System.out.println("붕어빵 이에요");
		ShowCount();
		System.out.println(flourkind);
	}
	void bbb() {
		System.out.println("bbb");
	}
}

class Animal260108{
	 void makeSound() {
		 System.out.println("으르렁");
	 }
}
class Dog260108 extends Animal260108{
	void makeSound() {
		 System.out.println("멍멍");
	 }
}
class Cat260108 extends Animal260108{
	void makeSound() {
		 System.out.println("야옹");
	 }
}


public class Java260108 {

	public static void main(String[] args) {
		// override
		/* 업케스팅
		 * 타입은 부모, new 는 자식.
		 * 속성은(변수)는 타입 따라가고,
		 * method는 override 따라가요
		 * */
		List<Bread260108> list1=new ArrayList<Bread260108>();
		Bread260108 fishbread =new FishBread260108("꿀가루");
		Bread260108 fishbread2 =new FishBread260108("흑가루");
		list1.add(fishbread);
		list1.add(fishbread2);
		list1.get(0).count=10;
		
		
		List<Animal260108> animals=new ArrayList<Animal260108>();
		Animal260108 dog=new Dog260108();
		Animal260108 cat=new Cat260108();
		Animal260108 animal=new Animal260108();
		animals.add(animal);
		animals.add(dog);
		animals.add(cat);
		
		for(Animal260108 e:animals) {
			e.makeSound();
		}
		
	}

}
