package com.javatest;

public class Java251229 {

	public static void main(String[] args) {
		//  >> << | &
		int num1=3;
		/*
		 * >> , << 요건 shift bit 연산 이라고 함
		 * 먼저 데이터를 2진수로 표현하고,
		 * << 이 방향은 0 으로 채우기
		 * >> 방향은 비트 떨구기
		 * */
		int num2= num1>>1;
		
		/*
		 * 함수: 기계. 기능을 수행함
		 * 껍데기가 있음.
		 * 재료를 넣는 투입구가 있음.
		 * 퉤 뱉는 출구도 있음.
		 * System.out.println() 요놈이 하는일은 화면에 출력하기
		 * */
		
		
		num1=3;
		num2=num1<<2;
		
		/* | & ^ */
		num1=2;
		num2=3;
		/*
		 * 연산자 우선순위-
		 * https://dojang.io/mod/page/view.php?id=188
		 * */
		int num3= num1 ^ num2 >> 1;
		System.out.println(num3);
	
	}

}
